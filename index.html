Q #1: Find the smallest number in an array.
   Input: [9, 3, 4, 2, 7]
   Output: 2
A: Walkthrough: Start with the first array value as the current minimum. Scan the array, replace the minimum whenever a smaller value is found, then print the final minimum.

class SmallestNumberInAnArray {
    public static void main(String[] args) {
        int [] num = {9,3,4,2,7};
        int len = num.length;
        int tem = num[0];
        for(int i=0; i < len; i++){
            if(num[i] < tem){
                tem=num[i];
            }
        }
        System.out.println("Smallest number in an array - "+tem);
    }
}
Q #2: Sort an integer array without using a built-in sorting function.
   Input: [9, 3, 4, 2, 7]
   Output: 2,3,4,7,9
A: Walkthrough: For each array position, compare its value with every value to its right. Swap whenever a smaller value is found; after all passes, print the values in ascending order.

class SortInteger {
    public static void main(String[] args) {
        int[] num = {9,3,4,2,7};
        int len = num.length;
        int temp = 0;
        for (int i = 0; i < len-1; i++) {
            for (int j = i + 1; j < len; j++) {
                if (num[i] > num[j]) {
                    temp = num[i];
                    num[i] = num[j];
                    num[j] = temp;
                    }
                }
            }
        for (int n: num)
            System.out.print(n + " ");
        }
    }

Q #5: Find two indexes whose values add up to a target number.
   Input  #1: nums = [2,7,11,15], target = 9
   Output #1: [0, 1]
   Input  #2: nums = [3,2,4], target = 6
   Output #2: [1,2]
A: Walkthrough: Try every pair of distinct indices with nested loops. Return the first pair whose values sum to the target; return [-1, -1] if no pair matches.

class ArrayOfTwoIndex{
    public static int[] indexOf(int [] nums, int target){
        int len = nums.length;
        for(int i=0;i<len;i++){
            for(int j=i+1;j<len;j++){
                if(nums[i]+nums[j]==target){
                return new int[]{i,j};
                }
            }
        }
    return new int[]{-1,-1};
}
    public static void main(String[] args) {
        int [] nums = {2,7,11,15};
        int target = 9;
        System.out.println(java.util.Arrays.toString(indexOf(nums, target)));
    }
}
Q #6: Remove duplicates from a sorted array in place.
   Input: [1, 2, 2, 2, 3, 5]
   Output: [1, 2, 3, 5]
A: Walkthrough: Because the array is sorted, duplicates are adjacent. Copy each new value into the next position in the same array, track the unique count, then print only the compacted prefix.

import java.util.Arrays;
class RemoveDuplicatesFromSortedArray {
    public static void main(String[] args) {
        int[] nums = {1, 2, 2, 2, 3, 5};
        int uniqueCount = 0;
        for (int num : nums) {
            if (uniqueCount == 0 || nums[uniqueCount - 1] != num) {
                nums[uniqueCount++] = num;
            }
        }
        System.out.println(Arrays.toString(Arrays.copyOf(nums, uniqueCount)));
    }
}

Q #8: Remove duplicates from an unsorted array without using a set.
   Input: [3, 1, 3, 2, 1]
   Output: [3, 1, 2]
A: Walkthrough: Keep distinct values in a separate array. For each input value, scan the values already collected and append it only if it has not appeared before; print the populated prefix.

import java.util.Arrays;
class RemoveDuplicatesFromUnsortedArray {
    public static void main(String[] args) {
        int[] nums = {3, 1, 3, 2, 1};
        int[] uniqueNumbers = new int[nums.length];
        int uniqueCount = 0;

        for (int num : nums) {
            boolean alreadyPresent = false;
            for (int i = 0; i < uniqueCount; i++) {
                if (uniqueNumbers[i] == num) {
                    alreadyPresent = true;
                    break;
                }
            }
            if (!alreadyPresent) {
                uniqueNumbers[uniqueCount++] = num;
            }
        }
        System.out.println(Arrays.toString(Arrays.copyOf(uniqueNumbers, uniqueCount)));
    }
}

Q #10: Move all zeros to the end of an array while preserving the order of the other values.
   Input: [0, 1, 0, 3, 12]
   Output: [1, 3, 12, 0, 0]
A: Walkthrough: Copy every nonzero value forward while preserving its order. Once they are gathered at the beginning, fill the remaining array positions with zeros.

import java.util.Arrays;
class MoveAllZerosToEndOfArray {
    public static void main(String[] args) {
        int[] num = {0, 1, 0, 3, 12};
        int nextNonZero = 0;
        for (int value : num) {
            if (value != 0) {
                num[nextNonZero++] = value;
            }
        }
        while (nextNonZero < num.length) {
            num[nextNonZero++] = 0;
        }
        System.out.println(Arrays.toString(num));
    }
}
Q #11: Determine whether a string is a palindrome.
   Input: "madam"
   Output: String is palindrome
A: Walkthrough: Compare each character with its mirrored character at the opposite end of the string. If any pair differs, mark the string as not a palindrome; otherwise it reads the same in both directions.

class StringPalindrom {
    public static void main(String[] args) {
        String value = "madam";
        int length=value.length();
        boolean isPalindrome=true;
        for(int i=0;i<length;i++){
            if(value.charAt(i)!=value.charAt(length-1-i)){
            isPalindrome=false;
        }
    }
    String result = isPalindrome? value+" string is palindrome" : value+" string is not palidrome";
    System.out.println(result);
    }
}

Q #13: Count each run of consecutive repeated characters in a string.
   Input: "AABBCABCGG"
   Output: A:2 B:2 C:1 A:1 B:1 C:1 G:2
A: Walkthrough: At each index, remember the current character and count how many identical characters appear consecutively. Print that character and run length, then continue at the next run.

class CountConsecutiveCharacters {
    public static void main(String[] args) {
        String str = "AABBCABCGG";
        int index = 0;
        while (index < str.length()) {
            char current = str.charAt(index);
            int count = 0;
            while (index < str.length() && str.charAt(index) == current) {
                count++;
                index++;
            }
            System.out.print(current + ":" + count + " ");
        }
    }
}

Q #15: Find the second-largest distinct number in an unsorted array.
   Input: [3, 2, 5, 1, 6]
   Output: 5
A: Walkthrough: Track the largest and second-largest distinct values in one pass. When a new maximum appears, move the old maximum into second place; otherwise update second place if the value is distinct and larger.

class SecondLargestArray {
    public static void main(String[] args) {
        int[] num = {3, 2, 5, 1, 6};
        Integer largest = null;
        Integer secondLargest = null;

        for (int value : num) {
            if (largest == null || value > largest) {
                secondLargest = largest;
                largest = value;
            } else if (value != largest
                    && (secondLargest == null || value > secondLargest)) {
                secondLargest = value;
            }
        }
        if (secondLargest == null) {
            System.out.println("No second-largest distinct value");
        } else {
            System.out.println(secondLargest);
        }
    }
}

Q #17: Count the occurrences of each word in a string, distinguishing uppercase and lowercase.
   Input: "This this is is done by Saket Saket is Saket"
   Output: {This=1, this=1, is=3, done=1, by=1, Saket=3}
A: Walkthrough: Trim the input and split it into words on whitespace. For each word, increment its count in a LinkedHashMap, which keeps first-seen order and treats differently cased words separately.

import java.util.LinkedHashMap;

class NumberOfWordsInString {
    public static void main(String[] args) {
        String str = "This this is is done by Saket Saket is Saket";
        String[] split = str.trim().split("\\s+");
        LinkedHashMap hashMap = new LinkedHashMap<>();
        for (String word : split) {
            hashMap.put(word, hashMap.getOrDefault(word, 0) + 1);
        }
        System.out.println(hashMap);
    }
}

Q #19: Shuffle a string so the character at position i moves to indices[i].
Example :1
   Input : s= "codeleet", indices=[4,5,6,7,0,2,1,3]
   Output: leetcode
Example :2
   Input : s= "abc", indices=[0,1,2]
   Output: abc
A: Walkthrough: Create a character array for the output. For each input position i, place that character at indices[i], then convert the filled array back into a string.

class ShufflingString {
    public static String shuffleString(String s, int[] indices) {
        char[] shuffled = new char[s.length()];
        for (int i = 0; i < s.length(); i++) {
            shuffled[indices[i]] = s.charAt(i);
        }
        return new String(shuffled);
    }

    public static void main(String[] args) {
        String s = "codeleet";
        int[] indices = {4, 5, 6, 7, 0, 2, 1, 3};
        System.out.println(shuffleString(s, indices));
    }
}
Q #20: Replace the first character with the last character in every word.
   Input: "hello world Java"
   Output: "oellh dorlw aavJ"
A: Walkthrough: Split the string into words, then swap the first and last characters of each word. Words with one character remain unchanged, and spaces between words are preserved.

class ReplaceFirstAndLastCharacter {
    public static String replaceFirstAndLast(String text) {
        String[] words = text.split(" ", -1);
        StringBuilder result = new StringBuilder();

        for (int wordIndex = 0; wordIndex < words.length; wordIndex++) {
            char[] characters = words[wordIndex].toCharArray();
            if (characters.length > 1) {
                char first = characters[0];
                characters[0] = characters[characters.length - 1];
                characters[characters.length - 1] = first;
            }

            if (wordIndex > 0) {
                result.append(' ');
            }
            result.append(characters);
        }

        return result.toString();
    }

    public static void main(String[] args) {
        String input = "hello world Java";
        System.out.println(replaceFirstAndLast(input));
    }
}
Q #21: Convert the first character of every word to uppercase.
   Input: "java programming is fun"
   Output: "Java Programming Is Fun"
A: Walkthrough: Walk through the string once. Convert a character to uppercase when it is the first character or follows a space; append every other character unchanged so the original spaces are retained.

class CapitalizeFirstCharacterOfEveryWord {
    public static String capitalizeWords(String text) {
        StringBuilder result = new StringBuilder(text.length());

        for (int i = 0; i < text.length(); i++) {
            char current = text.charAt(i);
            if (i == 0 || text.charAt(i - 1) == ' ') {
                result.append(Character.toUpperCase(current));
            } else {
                result.append(current);
            }
        }

        return result.toString();
    }

    public static void main(String[] args) {
        String input = "java programming is fun";
        System.out.println(capitalizeWords(input));
    }
}
Q #22: Rotate a number to the left by two digits.
   Input: 123456
   Output: 345612
A: Walkthrough: Count the digits, then extract them from right to left into an integer array so they are stored in their original order. Print from index two to the end, followed by the first two digits. This handles nonnegative integers and rotates by two positions, or fewer when the number has fewer than two digits.

class RotateNumberLeft {
    public static void main(String[] args) {
    int number = 123456;
    int temp = number;
    int digitCount = 0;

    do {
        digitCount++;
        temp /= 10;
    } while (temp > 0);

    int[] digits = new int[digitCount];
    temp = number;
    for (int i = digitCount - 1; i >= 0; i--) {
        digits[i] = temp % 10;
        temp /= 10;
    }

    int rotation = 2 % digitCount;
    for (int i = rotation; i < digitCount; i++) {
        System.out.print(digits[i]);
    }
    for (int i = 0; i < rotation; i++) {
        System.out.print(digits[i]);
    }
    System.out.println();
    }
}
Q #23: Rotate a number to the right by two digits.
   Input: 123456
   Output: 561234
A: Walkthrough: Count the digits, then extract them from right to left into an integer array so they are stored in their original order. Print the final two digits first, followed by the remaining digits. This handles nonnegative integers and rotates by two positions, or fewer when the number has fewer than two digits.

class RotateNumberRight {
    public static void main(String[] args) {
    int number = 123456;
    int temp = number;
    int digitCount = 0;

    do {
        digitCount++;
        temp /= 10;
    } while (temp > 0);

    int[] digits = new int[digitCount];
    temp = number;
    for (int i = digitCount - 1; i >= 0; i--) {
        digits[i] = temp % 10;
        temp /= 10;
    }

    int rotation = 2 % digitCount;
    int splitIndex = digitCount - rotation;
    for (int i = splitIndex; i < digitCount; i++) {
        System.out.print(digits[i]);
    }
    for (int i = 0; i < splitIndex; i++) {
        System.out.print(digits[i]);
    }
    System.out.println();
    }
